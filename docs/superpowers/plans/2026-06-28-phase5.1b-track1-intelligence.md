# Phase 5.1b — Track 1 Intelligence Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the frozen Evidence Pack into a scored Track 1 (Supplier Identity) signal: the LLM proposes ADR-G003 `weight_key` classifications over the pack; a deterministic six-gate validation firewall decides what scores; `deriveTrackSignal` (unchanged) scores only validated evidence; then the case reaches a verdict and writes institutional memory.

**Architecture:** `runTrack1` orchestrates acquisition (5.1a) → persists the pack/metrics → assigns stable source ids → calls the LLM (pack-interpretation prompt) → runs the firewall (`weightValidation.ts`) → builds validated `EvidenceItem`s + an audit trail → returns a `TrackOutput`. The firewall and gate config are pure deterministic code (the `weights.ts`/`source_profile.ts` pattern). Nothing changes `deriveTrackSignal`/`computeVerdict`/`weights.ts` scoring — the firewall is additive **upstream**.

**Tech Stack:** TypeScript, Next.js 16, vitest, Supabase, `runModel`→Anthropic (Sonnet 4.6, temp 0).

**Spec:** `docs/superpowers/specs/2026-06-27-phase5.1b-track1-intelligence-design.md` + the 2026-06-28 consolidated prompt (3 modifications).

```ts
import type { TrackKey } from "@/lib/constants/tracks";
import type { SourceProfile, AuthorityScore } from "@/lib/research/source_profile";
import { authorityFor } from "@/lib/research/source_profile";
import { weightFor, weightKeyExistsInAnyTrack, alternativeGroupFor } from "@/lib/research/weights";
import type { WeightValidation, ValidationGate, RejectionReason } from "@/lib/research/contracts";

// ── Gate config (code-owned trust rules; same pattern as weights.ts / source_profile.ts) ──
// Provenance gate (②): which source_profiles may earn each Track 1 weight_key.
const ALLOWED_PROFILES: Record<string, SourceProfile[]> = {
  government_registration: ["government_record", "registry"],
  domain_age_5_plus: ["whois"], domain_age_2_5: ["whois"], domain_age_under_2_established: ["whois"],
  address_verifiable: ["government_record", "registry", "official_company"],
  linkedin_company: ["social", "official_company"],
  phone_verifiable: ["official_company", "registry", "government_record"],
  website_quality: ["official_company", "official_brand"],
  bbb_or_trade_association: ["registry"],
  negative_reputation: ["forum", "social", "news", "marketplace"],
  registration_fabricated: ["government_record", "registry"],
  address_fraudulent: ["government_record", "registry", "news"],
  website_fraudulent: ["news", "forum", "marketplace"],
  scam_reports_corroborated: ["forum", "social", "news", "marketplace"],
};
// Authority gate (③): minimum authority for each key (defense-in-depth; mostly implied by ②).
const MIN_AUTHORITY: Record<string, AuthorityScore> = {
  government_registration: "high", domain_age_5_plus: "high", domain_age_2_5: "high",
  domain_age_under_2_established: "high", address_verifiable: "medium", linkedin_company: "low",
  phone_verifiable: "low", website_quality: "low", bbb_or_trade_association: "low",
  negative_reputation: "low", registration_fabricated: "high", address_fraudulent: "medium",
  website_fraudulent: "low", scam_reports_corroborated: "low",
};
const AUTH_RANK: Record<AuthorityScore, number> = { low: 0, medium: 1, high: 2 };

export interface ProposedMapping {
  evidence_id: string;
  proposed_weight_key: string;
  cited_source_ids: string[];
}
export interface FirewallInput {
  track: TrackKey;
  proposals: ProposedMapping[];
  sourceProfileById: Record<string, SourceProfile>;
}

const reject = (p: ProposedMapping, gate: ValidationGate | null, reason: RejectionReason): WeightValidation =>
  ({ evidence_id: p.evidence_id, proposed_weight_key: p.proposed_weight_key, validated_weight_key: null, gate, rejection_reason: reason });

export function validateWeights(input: FirewallInput): WeightValidation[] {
  const { track, proposals, sourceProfileById } = input;

  // ── Per-item gates: UNKNOWN → ① registry → ④ track → ⑥ grounding → ② provenance → ③ authority ──
  type Pass = { v: WeightValidation; profile: SourceProfile; sourceId: string; hardFail: boolean };
  const passed: Pass[] = [];
  const out: WeightValidation[] = [];

  for (const p of proposals) {
    if (p.proposed_weight_key === "UNKNOWN") { out.push(reject(p, null, "llm_returned_unknown")); continue; }
    if (!weightKeyExistsInAnyTrack(p.proposed_weight_key)) { out.push(reject(p, "registry", "registry")); continue; }
    const w = weightFor(track, p.proposed_weight_key);
    if (!w) { out.push(reject(p, "track", "track")); continue; }                       // exists, wrong track
    const cited = p.cited_source_ids.filter((id) => id in sourceProfileById);
    if (cited.length === 0) { out.push(reject(p, "grounding", "no_valid_citation")); continue; }
    // primary cited source = highest authority among cited
    const sourceId = cited.reduce((best, id) =>
      AUTH_RANK[authorityFor(sourceProfileById[id])] > AUTH_RANK[authorityFor(sourceProfileById[best])] ? id : best);
    const profile = sourceProfileById[sourceId];
    if (!(ALLOWED_PROFILES[p.proposed_weight_key] ?? []).includes(profile)) { out.push(reject(p, "provenance", "provenance")); continue; }
    if (AUTH_RANK[authorityFor(profile)] < AUTH_RANK[MIN_AUTHORITY[p.proposed_weight_key] ?? "low"]) { out.push(reject(p, "authority", "authority")); continue; }
    passed.push({
      v: { evidence_id: p.evidence_id, proposed_weight_key: p.proposed_weight_key, validated_weight_key: p.proposed_weight_key, gate: null, rejection_reason: null },
      profile, sourceId, hardFail: !!w.hard_fail,
    });
  }

  // ── Cross-item pass ⑤: dedupe, then contradictions ──
  // Dedupe identical (same key + same source) — keep the first.
  const seen = new Set<string>();
  const deduped: Pass[] = [];
  for (const x of passed) {
    const k = `${x.v.validated_weight_key} ${x.sourceId}`;
    if (seen.has(k)) { out.push(reject({ evidence_id: x.v.evidence_id, proposed_weight_key: x.v.proposed_weight_key, cited_source_ids: [x.sourceId] }, "contradiction", "contradiction")); continue; }
    seen.add(k); deduped.push(x);
  }

  // (b) hard_fail vs positive on the SAME source → hard_fail wins.
  const bySource = new Map<string, Pass[]>();
  for (const x of deduped) bySource.set(x.sourceId, [...(bySource.get(x.sourceId) ?? []), x]);
  const killed = new Set<string>(); // evidence_id rejected by contradiction
  for (const group of bySource.values()) {
    if (group.some((x) => x.hardFail) && group.some((x) => !x.hardFail)) {
      for (const x of group) if (!x.hardFail) { killed.add(x.v.evidence_id); out.push(reject({ evidence_id: x.v.evidence_id, proposed_weight_key: x.v.proposed_weight_key, cited_source_ids: [x.sourceId] }, "contradiction", "contradiction")); }
    }
  }

  // (a/c) mutually-exclusive buckets across DIFFERENT sources.
  const survivors = deduped.filter((x) => !killed.has(x.v.evidence_id));
  const byGroup = new Map<string, Pass[]>();
  for (const x of survivors) {
    const g = alternativeGroupFor(x.v.validated_weight_key!);
    if (!g) continue;
    const gid = g.join("|");
    byGroup.set(gid, [...(byGroup.get(gid) ?? []), x]);
  }
  for (const members of byGroup.values()) {
    if (members.length < 2) continue;
    const ranks = members.map((m) => AUTH_RANK[authorityFor(m.profile)]);
    const max = Math.max(...ranks);
    const top = members.filter((m) => AUTH_RANK[authorityFor(m.profile)] === max);
    if (top.length === 1) {
      // ⑤a — unequal authority: the single top survives, the rest are rejected.
      for (const m of members) if (m !== top[0]) { killed.add(m.v.evidence_id); out.push(reject({ evidence_id: m.v.evidence_id, proposed_weight_key: m.v.proposed_weight_key, cited_source_ids: [m.sourceId] }, "contradiction", "contradiction")); }
    } else {
      // ⑤c — equal authority: reject ALL tied members (disagreement is itself a finding).
      for (const m of top) { killed.add(m.v.evidence_id); out.push(reject({ evidence_id: m.v.evidence_id, proposed_weight_key: m.v.proposed_weight_key, cited_source_ids: [m.sourceId] }, "contradiction", "contradiction_equal_authority")); }
    }
  }

  for (const x of deduped) if (!killed.has(x.v.evidence_id)) out.push(x.v);
  return out;
}
```

```ts
import type { TrackContext } from "@/lib/research/contracts";
import type { ResearchQuestion } from "@/lib/research/acquisition/types";

function host(website: string | null): string | null {
  if (!website) return null;
  try { return new URL(website.startsWith("http") ? website : `https://${website}`).hostname; } catch { return null; }
}

// Code-defined Track 1 query set (the coverage lever). phone_verifiable + contact consistency are NOT
// separate queries — the LLM surfaces them from the registry/address/LinkedIn/BBB results.
export function buildTrack1Requests(ctx: TrackContext): { question: ResearchQuestion; input: string }[] {
  const v = ctx.vendor_name ?? "";
  const h = host(ctx.vendor_website);
  const reqs: { question: ResearchQuestion; input: string }[] = [
    { question: "business_registry", input: `${v} business registration secretary of state` },
    { question: "linkedin_presence", input: `${v} LinkedIn company` },
    { question: "bbb_listing", input: `${v} BBB better business bureau` },
    { question: "address_verification", input: `${v} address phone contact` },
    { question: "scam_reports", input: `${v} scam complaints reviews fraud` },
  ];
  if (h) reqs.unshift({ question: "domain_age", input: h });
  return reqs;
}
```

```ts
const CERTAINTIES = new Set(["verified", "inferred", "unknown"]);
export function parseTrack1Output(json: unknown): ParsedTrack1 {

  if (o._parse_error || !Array.isArray(o.evidence_items)) {
    return { items: [], reasoning_notes: "could not parse model output", unknowns: [] };
  }
  const items: ProposedEvidenceItem[] = [];
  for (const raw of o.evidence_items as Record<string, unknown>[]) {
    if (typeof raw?.proposed_weight_key !== "string" || typeof raw?.evidence_id !== "string") continue;
    items.push({
      evidence_id: raw.evidence_id,
      statement: typeof raw.statement === "string" ? raw.statement : "",
      proposed_weight_key: raw.proposed_weight_key,
      supporting_source_ids: Array.isArray(raw.supporting_source_ids) ? raw.supporting_source_ids.filter((x): x is string => typeof x === "string") : [],
      mapping_justification: typeof raw.mapping_justification === "string" ? raw.mapping_justification : "",
      certainty: CERTAINTIES.has(raw.certainty as string) ? (raw.certainty as ProposedEvidenceItem["certainty"]) : "unknown",
    });
  }
  return {
    items,
    reasoning_notes: typeof o.reasoning_notes === "string" ? o.reasoning_notes : "",
    unknowns: Array.isArray(o.unknowns) ? (o.unknowns as Unknown[]) : [],
  };
}
```

```ts
import type { TrackContext, TrackOutput, EvidenceItem } from "@/lib/research/contracts";
import { Orchestrator } from "@/lib/research/acquisition/orchestrator";
import { whoisPlugin } from "@/lib/research/acquisition/plugins/whois";
import { serperPlugin } from "@/lib/research/acquisition/plugins/serper";
import { nativeWebSearchPlugin } from "@/lib/research/acquisition/plugins/nativeWebSearch";
import { persistEvidencePack, persistAcquisitionMetrics } from "@/lib/data/acquisition";
import { runModel } from "@/lib/ai/runModel";
import { buildTrack1Requests } from "@/lib/research/tracks/track1.queries";
import { buildTrack1Prompt, parseTrack1Output } from "@/lib/research/track1.prompt";
import { validateWeights } from "@/lib/research/weightValidation";
import type { RawSource } from "@/lib/research/acquisition/types";

// Track 1 — Supplier Identity. Acquire (5.1a) → interpret (LLM proposes) → validate (firewall) →
// validated evidence_items + audit. The orchestrator acquires; the AI interprets; code decides.
export async function runTrack1(ctx: TrackContext): Promise<TrackOutput> {
  const requests = buildTrack1Requests(ctx);
  const orchestrator = new Orchestrator([whoisPlugin, serperPlugin, nativeWebSearchPlugin]);
  const { pack, metrics } = await orchestrator.gather({ case_id: ctx.case_id, track_key: "supplier_identity", requests });
  await persistEvidencePack(pack);
  await persistAcquisitionMetrics(ctx.case_id, "supplier_identity", metrics);

  // Stable source ids for citation; maps for provenance attach + firewall profile lookup.
  const byId = new Map<string, RawSource>();
  const sourceProfileById: Record<string, import("@/lib/research/source_profile").SourceProfile> = {};
  const promptSources = pack.sources.map((s, i) => {
    const id = `src_${i}`;
    byId.set(id, s);
    sourceProfileById[id] = s.provenance.source_profile;
    return { source_id: id, url: s.url, title: s.title, snippet: s.snippet };
  });

  const { system, user } = buildTrack1Prompt({ vendor_name: ctx.vendor_name, vendor_website: ctx.vendor_website }, promptSources);
  let parsed;
  try {
    const res = await runModel({ task: "track", system, user, temperature: 0 });
    parsed = parseTrack1Output(res.json);
  } catch {
    parsed = parseTrack1Output({ _parse_error: true });
  }

  const validations = validateWeights({
    track: "supplier_identity",
    sourceProfileById,
    proposals: parsed.items.map((it) => ({ evidence_id: it.evidence_id, proposed_weight_key: it.proposed_weight_key, cited_source_ids: it.supporting_source_ids })),
  });
  const validatedById = new Map(validations.map((v) => [v.evidence_id, v]));

  const evidence_items: EvidenceItem[] = [];
  for (const it of parsed.items) {
    const v = validatedById.get(it.evidence_id);
    if (!v || !v.validated_weight_key) continue; // rejected → audit only, never scores
    const primaryId = it.supporting_source_ids.find((id) => byId.has(id));
    const src = primaryId ? byId.get(primaryId) : undefined;
    evidence_items.push({
      evidence_id: it.evidence_id,
      statement: it.statement,
      certainty: it.certainty,
      source_type: src?.provenance.source_type ?? "inference",
      source_url: src?.url ?? null,
      claimant: "independent_registry",
      claimant_benefits: false,
      supports: "supplier_identity",
      weight_key: v.validated_weight_key,            // VALIDATED key only
      provenance: src?.provenance,                    // full provenance chain
    });
  }

  return {
    track_key: "supplier_identity",
    evidence_items,
    evidence_weights_applied: [],
    reasoning_notes: parsed.reasoning_notes,
    unknowns: parsed.unknowns,
    weight_validation: validations,
  };
}
```
